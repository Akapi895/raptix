  const selector = page.getByRole('combobox', { name: 'Bound thread' })
  await expect(selector).not.toHaveValue('', { timeout: 60000 })
  const listed = await (await request.get(`${api}/engagements/${project.id}/threads`)).json()

  const approvalsDir = path.join(process.env.W11_TEST_WORKSPACE!, slug, 'approvals')

async function cypher(statement: string, parameters: Record<string, unknown>): Promise<{ status: number; rows: unknown[] }> {
  const response = await fetch('http://127.0.0.1:7475/db/neo4j/tx/commit', {
    method: 'POST',
    headers: { 'content-type': 'application/json', authorization: `Basic ${Buffer.from(`neo4j:${process.env.W13_NEO4J_PASSWORD}`).toString('base64')}` },
    body: JSON.stringify({ statements: [{ statement, parameters }] }),
    signal: AbortSignal.timeout(10000),
  })
  const body = await response.json().catch(() => null) as { results?: Array<{ data?: Array<{ row?: unknown[] }> }> } | null
  return { status: response.status, rows: body?.results?.[0]?.data?.[0]?.row ?? [] }
}

import { randomBytes } from 'node:crypto'
import { readFileSync, renameSync, statSync, writeFileSync } from 'node:fs'
import { resolve } from 'node:path'

// Explicit, one-time bootstrap. The key stays in the operator's ignored .env,
// is never printed, and an existing valid key is never rotated implicitly.
const path = resolve(process.argv[2] ?? '.env')
const original = readFileSync(path, 'utf8')
const entries = original.match(/^VAULT_ENCRYPTION_KEY=.*$/gm) ?? []
if (entries.length > 1) throw Error('Duplicate VAULT_ENCRYPTION_KEY entries; resolve them before configuring the vault')
const raw = entries[0]?.slice('VAULT_ENCRYPTION_KEY='.length).trim().replace(/^(['"])(.*)\1$/, '$2') ?? ''
if (raw) {
  const key = Buffer.from(raw, 'base64')
  if (key.length !== 32 || key.toString('base64') !== raw) throw Error('Existing VAULT_ENCRYPTION_KEY is invalid; it has not been changed')
  console.log('Existing valid vault key retained; no rotation performed')
} else {
  const entry = `VAULT_ENCRYPTION_KEY=${randomBytes(32).toString('base64')}`
  const newline = original.includes('\r\n') ? '\r\n' : '\n'
  const updated = entries.length ? original.replace(/^VAULT_ENCRYPTION_KEY=.*$/m, entry) : `${original}${original.endsWith('\n') ? '' : newline}${entry}${newline}`
  const temporary = `${path}.vault-init.tmp`
  writeFileSync(temporary, updated, { mode: statSync(path).mode, flag: 'wx' })
  renameSync(temporary, path)
  console.log('Persistent 32-byte vault key configured; value withheld')
}

"""Export the runtime-owned planning schemas for the independent API validator."""

import json
import sys
from pathlib import Path
from typing import Any

from raptix_core.types import engagement
from raptix_core.types.engagement import (
    CONOPS,
    OPPLAN,
    AbortPlan,
    CleanupPlan,
    ContactPlan,
    DataHandlingPlan,
    DeconflictionPlan,
    RoE,
    ThreatProfile,
)

MODELS = {
    "roe": RoE,
    "conops": CONOPS,
    "deconfliction": DeconflictionPlan,
    "opplan": OPPLAN,
    "threat-profile": ThreatProfile,
    "contact": ContactPlan,
    "data-handling": DataHandlingPlan,
    "abort": AbortPlan,
    "cleanup": CleanupPlan,
}


def schema_with_factory_defaults(model: Any) -> dict[str, Any]:
    schema = model.model_json_schema()
    sections = [(model, schema)] + [
        (getattr(engagement, name, None), definition)
        for name, definition in schema.get("$defs", {}).items()
    ]
    for owner, section in sections:
        for name, field in getattr(owner, "model_fields", {}).items():
            if field.default_factory is None:
                continue
            default = field.default_factory()
            if isinstance(default, (list, dict)):
                section["properties"][name]["default"] = json.loads(
                    json.dumps(default, default=lambda value: value.model_dump(mode="json"))
                )
    return schema


if __name__ == "__main__":
    destination = Path(__file__).resolve().parents[1] / "server/src/services/planning-schemas.json"
    exported = (
        json.dumps(
            {name: schema_with_factory_defaults(model) for name, model in MODELS.items()}, indent=2
        )
        + "\n"
    )
    if "--check" in sys.argv:
        if destination.read_text(encoding="utf-8") != exported:
            raise SystemExit("Planning schemas differ from the runtime models")
    else:
        destination.write_text(exported, encoding="utf-8")
